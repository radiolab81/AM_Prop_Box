cmake_minimum_required(VERSION 3.16)
project(amp_host_test CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AMP_DOUBLE "double statt float (trennt Portierungs- von Rundungsfehlern)" OFF)

set(PROP ${CMAKE_CURRENT_LIST_DIR}/../components/propagation)
file(GLOB SRCS ${PROP}/src/*.cpp)
add_executable(amp_test test_main.cpp ${SRCS})
target_include_directories(amp_test PRIVATE ${PROP}/include ${PROP}/src)
target_compile_options(amp_test PRIVATE -Wall -Wextra -Wshadow -Wconversion -Wno-sign-conversion)
if(AMP_DOUBLE)
  target_compile_definitions(amp_test PRIVATE AMP_DOUBLE)
endif()
